cmake_minimum_required(VERSION 3.24)
project(strata_split_v100_experimental LANGUAGES CXX CUDA)
if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.0 OR NOT CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13.0)
  message(FATAL_ERROR "The experimental V100 build needs CUDA 12.x; CUDA 13 no longer compiles sm_70")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The experimental V100 build is Linux only")
endif()
get_filename_component(STRATA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(STRATA_ENABLE_CUDA ON)
set(STRATA_BUILD_TESTS ON)
enable_testing()
set(STRATA_SPLIT_ARCH 70 CACHE STRING "V100 compute capability")
if(NOT STRATA_SPLIT_ARCH STREQUAL "70")
  message(FATAL_ERROR "This build recipe is only for V100 / sm_70")
endif()
set(STRATA_SPLIT_CARDS "2" CACHE STRING "Number of identical V100 cards")
if(NOT STRATA_GGML_DIR OR NOT EXISTS "${STRATA_GGML_DIR}/ggml/src/ggml-cuda/vecdotq.cuh")
  message(FATAL_ERROR "Set STRATA_GGML_DIR to the pinned llama.cpp checkout (third_party/ggml/VERSION.txt)")
endif()
# Independent MTP engine. No dependency on the normal engine or its DFlash2 library.
add_subdirectory("${STRATA_SOURCE_DIR}/src/split" split)
